cmake_minimum_required(VERSION 3.22)
project(NoJumpDelay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NOJUMP_TEST_ONLY "Build host-side tests only" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(nojump_tests tests/test_core.cpp)
  target_include_directories(nojump_tests PRIVATE include)
  add_test(NAME nojump_core COMMAND nojump_tests)
endif()
if(NOJUMP_TEST_ONLY)
  return()
endif()
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Build the mod with the Android NDK and arm64-v8a")
endif()
include(FetchContent)
FetchContent_Declare(preloader
  GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git
  GIT_TAG cc08b40a31471d3e2dedb267f5f17b28da2b0604)
FetchContent_MakeAvailable(preloader)
add_library(NoJumpDelay SHARED src/mod.cpp)
target_include_directories(NoJumpDelay PRIVATE include)
target_link_libraries(NoJumpDelay PRIVATE preloader nlohmann_json::nlohmann_json log dl)
target_link_options(NoJumpDelay PRIVATE "-Wl,-z,max-page-size=16384")
target_compile_options(NoJumpDelay PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
