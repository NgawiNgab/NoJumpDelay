cmake_minimum_required(VERSION 3.22)
project(NoJumpDelay LANGUAGES CXX)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Build with the Android NDK and -DANDROID_ABI=arm64-v8a")
endif()
set(PRELOADER_INCLUDE_DIR "" CACHE PATH "Directory containing pl/Mod.hpp")
set(PRELOADER_LIBRARY "" CACHE FILEPATH "Android arm64 libpreloader.so matching the launcher")
if(NOT EXISTS "${PRELOADER_INCLUDE_DIR}/pl/Mod.hpp" OR NOT EXISTS "${PRELOADER_LIBRARY}")
  message(FATAL_ERROR "Provide PRELOADER_INCLUDE_DIR and PRELOADER_LIBRARY from the matching Preloader SDK")
endif()
add_library(preloader SHARED IMPORTED)
set_target_properties(preloader PROPERTIES IMPORTED_LOCATION "${PRELOADER_LIBRARY}")
add_library(NoJumpDelay SHARED src/main.cpp src/ComponentLookup.cpp)
target_compile_features(NoJumpDelay PRIVATE cxx_std_20)
target_compile_options(NoJumpDelay PRIVATE -fno-rtti -fPIC -ffunction-sections -fdata-sections)
target_include_directories(NoJumpDelay PRIVATE "${PRELOADER_INCLUDE_DIR}")
target_link_libraries(NoJumpDelay PRIVATE preloader log android)
target_link_options(NoJumpDelay PRIVATE -Wl,--gc-sections -Wl,--hash-style=gnu -Wl,-z,max-page-size=16384)
